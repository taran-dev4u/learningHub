Python 3 Solution (Linear Time O(k) Space) 
from typing import List

class Solution:
    def getRow(self, rowIndex: int) -> List[int]:
        row = [1] * (rowIndex + 1)
        for k in range(1, rowIndex):
            row[k] = row[k - 1] * (rowIndex - k + 1) // k
        return row
